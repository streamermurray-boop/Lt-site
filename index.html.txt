function normalize(rows){
  PRODUCTS=[];
  for(var i=0;i<rows.length;i++){
    var o=rows[i];
    var title=o.Title||o.Name||o.title||o.name||'';
    if(!title) continue;

    var img=normalizeImg(o.ImageURL||o.Image||o.image||o.imageurl||'');
    var rawLink=o.Link||o.URL||o.Url||o.href||o.url||o['ACBuy Page']||o['ACBUY Page']||o['ACBUY']||o['Page']||o['Product Link']||'';

    var sec=o.Category||o.category||guessSection(title);

    // ✅ Ignore CSV price. Enforce your rules:
    var price=enforceCategoryPrice(sec, title);

    var href=appendRef(rawLink);
    PRODUCTS.push({Title:title,Image:img,Price:price,_href:href,Section:sec});
  }

  FEATURED=PRODUCTS.slice(0,10);
  renderTabs();
  renderFeatured();
  renderGrid();
}
var file=$('#file');
if(file){
  file.addEventListener('change',function(e){
    var f=e.target.files[0];
    if(!f){log('No file selected');return;}
    log('Reading '+f.name+' …');
    f.text().then(function(t){
      var rows=parseCSV(t);
      log('Parsed '+rows.length+' rows');
      normalize(rows);
    }).catch(function(err){
      console.error(err);
      log('Error reading file');
    });
    e.target.value='';
  });
}

function bootFromFixedCSV(){
  log('Loading catalog…');
  fetch(CSV_URL_PRIMARY,{mode:'cors',cache:'no-store'})
    .then(function(r){ if(!r.ok) throw new Error('Primary fetch failed: '+r.status); return r.text(); })
    .catch(function(){
      return fetch(CSV_URL_FALLBACK,{cache:'no-store'})
        .then(function(r){ if(!r.ok) throw new Error('Fallback fetch failed: '+r.status); return r.text(); });
    })
    .then(function(text){
      var rows=parseCSV(text);
      log('Parsed '+rows.length+' rows');
      normalize(rows);
    })
    .catch(function(err){
      console.error(err);
      log('Failed to load the catalog file.');
    });
}
