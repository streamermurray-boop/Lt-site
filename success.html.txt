<script>
(async function(){
  const params = new URLSearchParams(location.search);
  const session_id = params.get("session_id");
  const pill = document.getElementById("statusPill");

  if(!session_id){
    pill.textContent = "Missing session_id";
    return;
  }

  const res = await fetch(`/api/get-session?session_id=${encodeURIComponent(session_id)}`);
  if(!res.ok){
    pill.textContent = "Could not load order details";
    return;
  }
  const data = await res.json();

  document.getElementById("orderNumber").textContent = data.orderNumber || "LT-UNKNOWN";
  document.getElementById("itemTitle").textContent = data.title || "—";
  document.getElementById("supportEmail").textContent = data.supportEmail || "support@example.com";

  if(data.payment_status === "paid"){
    pill.textContent = "Payment received";
  } else {
    pill.textContent = `Payment status: ${data.payment_status || "unknown"}`;
  }
})();
</script>
